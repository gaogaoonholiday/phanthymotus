"""
单张图片端到端测试 — 不经过 ROS2，直接调用 EdgeFaceAdapter + FaceDatabase
"""
import sys, os, time, json

# edgeface_src 加入 path（和 face.py 插件中的逻辑一致）
EDGEFACE_SRC = "/work/edgeface_src"
if EDGEFACE_SRC not in sys.path:
    sys.path.insert(0, EDGEFACE_SRC)

from plugins.face import EdgeFaceAdapter, FaceDatabase, DEFAULT_MODEL_NAME

# ── 配置 ──
MODEL_NAME  = os.environ.get("FACE_MODEL", DEFAULT_MODEL_NAME)
MODEL_DIR   = os.environ.get("FACE_MODEL_DIR", "/models/face")
FACE_DB_DIR = os.environ.get("FACE_DB_DIR", "/workspace/face_db")
DEVICE      = os.environ.get("FACE_DEVICE", "cpu")
TEST_IMAGE  = os.environ.get("TEST_IMAGE", "/workspace/face_db/n000001/0001_01.jpg")

print(f"=== Face Recognition E2E Test ===")
print(f"Model:     {MODEL_NAME}")
print(f"Model dir: {MODEL_DIR}")
print(f"Face DB:   {FACE_DB_DIR}")
print(f"Device:    {DEVICE}")
print(f"Test img:  {TEST_IMAGE}")

# ── 1. 加载模型 ──
print(f"\n[1/4] Loading EdgeFace + MTCNN...")
t0 = time.time()
adapter = EdgeFaceAdapter(MODEL_NAME, MODEL_DIR, DEVICE)
t1 = time.time()
print(f"  Model loaded in {t1-t0:.2f}s")

# ── 2. 加载身份库 ──
print(f"\n[2/4] Loading face database...")
t0 = time.time()
face_db = FaceDatabase()
face_db.load_from_dir(FACE_DB_DIR, adapter)
t1 = time.time()
print(f"  DB loaded in {t1-t0:.2f}s ({face_db.count()} embeddings)")

if face_db.is_empty():
    print("  WARNING: face_db is empty! Matching will always return 'unknown'")

# ── 3. 推理测试图片 ──
print(f"\n[3/4] Detecting faces in {TEST_IMAGE}...")
import cv2
import numpy as np

# 模拟容器内的流程: JPEG bytes → cv2.imdecode → RGB
with open(TEST_IMAGE, 'rb') as f:
    jpeg_bytes = f.read()

t0 = time.time()
frame = cv2.imdecode(np.frombuffer(jpeg_bytes, np.uint8), cv2.IMREAD_COLOR)
H, W = frame.shape[:2]
rgb_frame = frame[:, :, ::-1]  # BGR → RGB

detections = adapter.detect_and_embed(rgb_frame)
t1 = time.time()
print(f"  Detection: {len(detections)} face(s) in {t1-t0:.3f}s")
print(f"  Image size: {W}x{H}")

# ── 4. 身份匹配 + 输出格式 ──
print(f"\n[4/4] Identity matching...")
faces = []
for det in detections:
    x1, y1, x2, y2 = det["bbox"]
    bbox_relative = [
        round(x1 / W, 6),
        round(y1 / H, 6),
        round(x2 / W, 6),
        round(y2 / H, 6),
    ]
    embedding = np.array(det["embedding"], dtype=np.float32)
    person_id, similarity = face_db.match(embedding, threshold=0.5)

    face_result = {
        "detect_confidence": round(det["confidence"], 4),
        "bbox_relative": bbox_relative,
        "identity": {
            "person_id": person_id,
            "confidence": round(similarity, 4),
        }
    }
    faces.append(face_result)
    print(f"  bbox: {bbox_relative}")
    print(f"  detect_conf: {face_result['detect_confidence']}")
    print(f"  person_id: {person_id}, sim: {similarity:.4f}")

result = {
    "faces": faces,
}

print(f"\n=== Result (JSON) ===")
print(json.dumps(faces, ensure_ascii=False, indent=2))

# ── FPS 测试 ──
print(f"\n=== FPS Test (10 iterations) ===")
times = []
for i in range(10):
    t0 = time.time()
    dets = adapter.detect_and_embed(rgb_frame)
    for d in dets:
        face_db.match(np.array(d["embedding"], dtype=np.float32), threshold=0.5)
    t1 = time.time()
    times.append(t1 - t0)

avg_time = sum(times[2:]) / len(times[2:])  # skip first 2 (warmup)
fps = 1.0 / avg_time if avg_time > 0 else 0
print(f"  Avg inference time: {avg_time*1000:.1f}ms")
print(f"  Estimated FPS: {fps:.1f}")
print(f"  GPU memory: ", end="")
try:
    import torch
    if torch.cuda.is_available():
        mem = torch.cuda.memory_allocated() / 1024 / 1024
        print(f"{mem:.1f} MB")
    else:
        print("CUDA not available")
except:
    print("N/A")

print(f"\n=== Test Complete ===")
PYEOF

"""
向 ROS2 话题发布一张测试图片 (CompressedImage)，然后监听输出话题
"""
import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy, DurabilityPolicy
from sensor_msgs.msg import CompressedImage
from std_msgs.msg import String
import sys, time, json

INPUT_TOPIC  = sys.argv[1] if len(sys.argv) > 1 else "/test/camera/rgb"
OUTPUT_TOPIC = f"{INPUT_TOPIC}/face"
IMAGE_PATH   = sys.argv[2] if len(sys.argv) > 2 else "/workspace/face_db/n000001/0001_01.jpg"

rclpy.init()
node = Node("face_test_pub")

# 接收结果
results = []
qos = QoSProfile(
    reliability=ReliabilityPolicy.BEST_EFFORT,
    history=HistoryPolicy.KEEP_LAST,
    depth=10,
    durability=DurabilityPolicy.VOLATILE,
)

def on_result(msg: String):
    data = json.loads(msg.data)
    print(f"\n=== Received face recognition result ===")
    print(json.dumps(data, indent=2, ensure_ascii=False))
    results.append(data)

node.create_subscription(String, OUTPUT_TOPIC, on_result, qos)

# 发布图片
pub = node.create_publisher(CompressedImage, INPUT_TOPIC, qos)

with open(IMAGE_PATH, 'rb') as f:
    jpeg_data = f.read()

msg = CompressedImage()
msg.header.stamp = node.get_clock().now().to_msg()
msg.format = "jpeg"
msg.data = jpeg_data

# 等订阅者就绪
time.sleep(2)
pub.publish(msg)
print(f"Published {len(jpeg_data)} bytes to {INPUT_TOPIC}")
print(f"Waiting for result on {OUTPUT_TOPIC}...")

# 等待结果（最多 15 秒）
start = time.time()
while time.time() - start < 15 and not results:
    rclpy.spin_once(node, timeout_sec=0.1)

if not results:
    print("ERROR: No result received within 15 seconds!")
else:
    print(f"\nSuccess! Got {len(results)} result(s).")

node.destroy_node()
rclpy.shutdown()
PYEOF

9. 常见问题排查
================================================================================

┌─ 问题: torch.cuda.is_available() 返回 False ─────────────────────────┐
│ 原因: 容器未使用 --runtime nvidia                                    │
│ 解决: docker run 必须加 --runtime nvidia                             │
│       L4T base image 的驱动库是空占位文件，只有 nvidia runtime        │
│       才会挂载宿主机的真实驱动库。                                    │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: ImportError: No module named 'timm' ─────────────────────────┐
│ 原因: Dockerfile 中 timm 安装失败或跳过                               │
│ 解决: docker exec -it face-test pip3 install --no-deps timm          │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: MTCNN 权重加载失败 ────────────────────────────────────────────┐
│ 原因: 容器中缺少 pnet.npy、rnet.npy 或 onet.npy                     │
│ 解决: 这些权重已随仓库提交，确认 edgeface_src 目录完整复制到镜像中   │
│       并确认 face.py 能导入 face_alignment.mtcnn                   │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: "Cannot uninstall 'PyYAML'" ──────────────────────────────────┐
│ 原因: JP5 (focal) 的 python3-yaml 是 distutils 包，pip 无法卸载      │
│ 解决: Dockerfile 已用 --ignore-installed 修复。                      │
│       如手动安装: pip3 install --ignore-installed pyyaml              │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: face db loaded: 0 persons ──────────────────────────────────┐
│ 原因: 身份库目录为空或路径不对                                       │
│ 检查: ls -la /workspace/face_db/                                     │
│       确保目录结构为 face_db/<personId>/<image>.jpg                  │
│       确保容器挂载了 /workspace/face_db                              │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: 检测不到人脸 (Detection: 0 face(s)) ────────────────────────┐
│ 排查: 1) 图片是否为正脸？                                             │
│       2) 人脸是否太小？MTCNN min_face_size 当前为 40px               │
│       3) 尝试降低阈值:                                               │
│          config.yaml 中 confidence 默认 0.5，可临时降到 0.3          │
│       4) 确认图片解码成功: cv2.imdecode 返回非 None                  │
│       5) MTCNN GPU 单张高分辨率图片约 2 秒，需关注超时和吞吐量       │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: person_id 总是 "unknown" ────────────────────────────────────┐
│ 排查: 1) face_db 是否为空？                                          │
│       2) similarity_threshold 是否太高？默认 0.5，可降到 0.3 测试    │
│       3) 身份库照片和测试照片是否为同一人？                          │
│       4) 身份库照片是否正脸清晰？                                    │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: Docker build 时 cv2 import 失败 (mat_wrapper circular) ──────┐
│ 原因: JP5.11 base image 的 cv2 __init__.py 有循环导入               │
│ 解决: Dockerfile 已用 glob-based shim 修复 (见 Dockerfile.jetson)    │
│       如手动修复，参考 Dockerfile.jetson 中的 cv2 修复段落           │
└──────────────────────────────────────────────────────────────────────┘

┌─ 问题: GPU 占用或显存过高 ─────────────────────────────────────────┐
│ 当前 MTCNN + EdgeFace 使用 CUDA；单进程约 reserved 333 MB。          │
│ 10 个容器估算约 3.3 GB，超过约 1.6 GB 目标。                        │
│ 排查: 1) 确认 fps 设置合理 (默认 3fps)                              │
│       2) 用 nvidia-smi 或 tegrastats 监控 GPU/显存:                 │
│          sudo tegrastats --interval 1000 | grep GR3D                │
│       3) 若资源不满足，恢复 config.yaml 的 device: cpu              │
└──────────────────────────────────────────────────────────────────────┘

    - FPS (监控指标)
    - 推理失败率 (监控指标)

- 仓库必须为 public
- 必须填个人 fork 地址，不能填 4paradigm/phanthymotus
- commit_id 必须是你在 fork 上推送的 commit
- 不要上传大于 1MB 的文件（模型在 juicefs 上下载）
